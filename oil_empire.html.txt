/* STATS */

function clamp(v){
    return Math.max(0,Math.min(100,v));
}

function apply(change){

    for(const key in change){
        state[key]=clamp(state[key]+change[key]);
    }

    updateStats();
}

function updateStats(){

    for(const key of Object.keys(state)){

        document.getElementById(key).textContent=
            Math.round(state[key]);

        document.getElementById(key+"bar").style.width=
            state[key]+"%";
    }
}

    t.textContent=message;

    document.body.appendChild(t);

    setTimeout(()=>t.remove(),2300);
}


/* EVENT */

function setEvent(title,text){

    document.getElementById("event").innerHTML=
        "<b>"+title+"</b><span>"+text+"</span>";
}


/* MAP HIGHLIGHT */

function highlight(country){

    document.querySelectorAll(".country")
        .forEach(x=>x.classList.remove("hot"));

    if(country){
        const el=document.getElementById(country);
        if(el)el.classList.add("hot");
    }
}


/* RENDER */

function renderMission(){

    const m=missions[mission];

    document.getElementById("year").textContent=m.year;
    document.getElementById("counter").textContent=
        "MISSION "+(mission+1)+" / "+missions.length;

    document.getElementById("tag").textContent=m.tag;
    document.getElementById("missionTitle").textContent=m.title;
    document.getElementById("missionText").textContent=m.text;

    const area=document.getElementById("missionArea");
    area.innerHTML="";

    highlight(null);

    if(m.mini==="budget"){
        renderBudget();
        return;
    }

    if(m.mini==="tanker"){
        renderTanker();
        return;
    }

    if(m.mini==="environment"){
        renderEnvironment();
        return;
    }

    m.choices.forEach((choice,index)=>{

        const button=document.createElement("button");

        button.className="choice";

        button.innerHTML=
            "<b>"+choice[0]+"</b>"+
            "<small>"+choice[1]+"</small>";

        button.onclick=()=>choose(index);

        area.appendChild(button);
    });

    if(m.year==="1908")highlight("iran");
    if(m.year==="1951")highlight("iran");
    if(m.year==="1960"){
        highlight("iraq");
        document.getElementById("iraq").classList.add("clickable");
    }
    if(m.year==="1973"){
        highlight("saudi");
        document.getElementById("saudi").classList.add("clickable");
    }
    if(m.year==="1990–1991"){
        highlight("kuwait");
    }
}


/* NORMAL CHOICE */

function choose(index){

    const m=missions[mission];
    const choice=m.choices[index];

    apply(choice[2]);

    toast(choice[0]);

    if(m.year==="1908"){
        setEvent(
            "1908 — Oil production begins",
            "Commercial production at Masjed Soleyman begins a new oil era."
        );
    }

    if(m.year==="1951"){
        setEvent(
            "1951 — Iranian nationalisation",
            "The struggle for control over petroleum becomes a major political issue."
        );
    }

    if(m.year==="1960"){
        setEvent(
            "1960 — OPEC",
            "Oil-producing countries create a new collective organisation."
        );
    }

    if(m.year==="1973"){
        setEvent(
            "1973 — Oil crisis",
            "The crisis demonstrates the strategic importance of petroleum."
        );
    }

    if(m.year==="1990–1991"){
        setEvent(
            "1990–91 — Gulf War",
            "Kuwait's invasion and the war place Gulf oil security at the centre of world politics."
        );
    }

    setTimeout(nextMission,700);
}


/* NEXT */

function nextMission(){

    mission++;

    if(mission>=missions.length){
        finishGame();
    }else{
        renderMission();
    }
}

function budgetUpdate(){

    const values=[
        +document.getElementById("in1").value,
        +document.getElementById("in2").value,
        +document.getElementById("in3").value,
        +document.getElementById("in4").value
    ];

    values.forEach((v,i)=>{
        document.getElementById("b"+(i+1)).textContent=v;
    });

    document.getElementById("total").textContent=
        values.reduce((a,b)=>a+b,0);
}

function finishBudget(){

    const values=[
        +in1.value,+in2.value,+in3.value,+in4.value
    ];

    const total=values.reduce((a,b)=>a+b,0);

    if(total!==100){
        toast("You must allocate exactly 100 points");
        return;
    }

    apply({
        wealth:Math.round(values[0]/8),
        security:Math.round(values[1]/8),
        influence:Math.round(values[2]/8),
        environment:-Math.round(values[3]/12)
    });

    setEvent(
        "1970s–80s — Oil wealth transforms society",
        "Oil revenue supports infrastructure, public services and modernisation."
    );

    toast("Development budget confirmed!");

    setTimeout(nextMission,900);
}

function moveTanker(type){

    if(routeStep>=5)return;

    routeStep++;

    document.getElementById("c"+(routeStep-1))
        .classList.add("done");

    const positions=[
        ["65%","70%"],
        ["59%","65%"],
        ["53%","61%"],
        ["47%","56%"],
        ["40%","51%"]
    ];

    const p=positions[routeStep-1];

    document.getElementById("tanker").style.left=p[0];
    document.getElementById("tanker").style.top=p[1];

    if(type===2){
        apply({security:3,wealth:-1});
        toast("Protected checkpoint reached");
    }

    if(type===1){
        apply({wealth:2,security:-1});
        toast("Fast route!");
    }

    if(type===-1){
        apply({security:-4,wealth:3,environment:1});
        toast("Dangerous shortcut!");
    }

    if(routeStep===5){

        setEvent(
            "1980s — Tanker War",
            "Oil shipping routes become security concerns during the Iran–Iraq War."
        );

        setTimeout(nextMission,1200);
    }
}

function finishGame(){

    document.getElementById("game").classList.add("hidden");
    document.getElementById("ending").classList.remove("hidden");

    document.getElementById("fwealth").textContent=Math.round(state.wealth);
    document.getElementById("foil").textContent=Math.round(state.oil);
    document.getElementById("finfluence").textContent=Math.round(state.influence);
    document.getElementById("fsecurity").textContent=Math.round(state.security);
    document.getElementById("fenv").textContent=Math.round(state.environment);

    let title;
    let text;

    if(state.environment<30){

        title="A SHIFT TOWARD SUSTAINABILITY";

        text=
        "Your decisions produced significant wealth and influence while keeping environmental pressure relatively low. The simulation shows how diversification, efficiency and environmental choices can change the long-term direction of an oil economy.";

    }else if(state.environment<60){

        title="A BALANCED OIL ECONOMY";

        text=
        "Your state entered 2000 with a mixture of oil wealth, development and environmental pressure. The choices show the difficult balance between economic growth, security and environmental concerns.";

    }else{

        title="AN OIL-DEPENDENT FUTURE";

        text=
        "Your state entered 2000 with strong petroleum wealth, but environmental pressure and dependence remained significant. The simulation demonstrates the long-term trade-offs created by an oil-based economy.";
    }

    document.getElementById("finalTitle").textContent=title;
    document.getElementById("finalText").textContent=text;
}
